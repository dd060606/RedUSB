REM Opens a terminal in an OS-specific way, then types a greeting.
REM The device guesses the host OS from how it enumerated the USB device.
REM Windows is detected reliably; Linux vs macOS is a best-effort guess.
DEFAULT_DELAY 40
IF_OS WINDOWS
  GUI r
  DELAY 400
  STRINGLN cmd
ELSE
  IF_OS MAC
    REM Spotlight -> Terminal
    GUI SPACE
    DELAY 400
    STRINGLN Terminal
  ELSE
    REM Linux or unknown: try the common terminal shortcut
    CTRL ALT t
  END_IF
END_IF
DELAY 800
STRINGLN echo Hello from RedUSB
